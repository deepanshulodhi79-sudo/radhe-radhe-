async function sendAll(){ const emails = validate(); if (!emails) return; const senderName = document.getElementById('senderName').value.trim(); const gmailId = document.getElementById('gmailId').value.trim(); const appPassword = document.getElementById('appPassword').value.trim(); const subject = document.getElementById('subject').value.trim(); const messageBody = document.getElementById('messageBody').value.trim(); const btn = document.getElementById('sendBtn'); btn.disabled = true; btn.innerHTML = '⏳ Dispatching Batch...'; clearLog(); setStatus('sending', '📤', `Processing ${emails.length} recipients...`); setProgress(50, 100); try { // Single High-Speed Batch Call const res = await fetch('/api/send-batch', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ senderName, gmailId, appPassword, subject, messageBody, recipients: emails }) }); const d = await res.json(); if (res.ok && d.success) { setProgress(100, 100); setStatus('success', '🎉', `Batch Complete! ${d.delivered} Delivered | ${d.failed} Failed`); addLog('ok', '✅', `Successfully sent ${d.delivered} emails`); if (d.failed > 0) addLog('fail', '❌', `${d.failed} emails failed`); } else { setStatus('error', '💥', d.message || 'Batch Dispatch Failed'); addLog('fail', '❌', d.message || 'Credential or Server Error'); } } catch (err) { setStatus('error', '💥', 'Server Connection Error'); addLog('fail', '❌', 'Failed to communicate with server'); } btn.disabled = false; btn.innerHTML = '🚀 Send All Emails'; }
